# CMakeLists in this exact order for cmake to work correctly
cmake_minimum_required(VERSION 3.16)

# First-run: generate .vscode/settings.json from template (idempotent)
if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/.vscode/settings.json"
   AND EXISTS "${CMAKE_CURRENT_LIST_DIR}/.vscode/settings.template.json")
    find_program(_setup_vscode_python NAMES python3 python)
    if(_setup_vscode_python)
        execute_process(
            COMMAND "${_setup_vscode_python}" "${CMAKE_CURRENT_LIST_DIR}/tools/setup_vscode.py"
            WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}"
        )
    endif()
endif()

# The firmware is written against ESP-IDF 5.5.x from 5.5.4 on. Older trees miss
# fixes it relies on and 6.x changed APIs it uses; say so here instead of
# failing somewhere inside a component.
set(CROSSPAD_IDF_MAJOR 5)
set(CROSSPAD_IDF_MINOR 5)
set(CROSSPAD_IDF_MIN_PATCH 4)
include($ENV{IDF_PATH}/tools/cmake/version.cmake)
if(NOT IDF_VERSION_MAJOR EQUAL CROSSPAD_IDF_MAJOR OR NOT IDF_VERSION_MINOR EQUAL CROSSPAD_IDF_MINOR
        OR IDF_VERSION_PATCH LESS CROSSPAD_IDF_MIN_PATCH)
    message(FATAL_ERROR
        "ESP-IDF ${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH} at $ENV{IDF_PATH}: "
        "CrossPad builds with ${CROSSPAD_IDF_MAJOR}.${CROSSPAD_IDF_MINOR}.x, "
        "${CROSSPAD_IDF_MAJOR}.${CROSSPAD_IDF_MINOR}.${CROSSPAD_IDF_MIN_PATCH} or newer "
        "(https://crosspad.app/docs/how-to/esp-idf-versions/)")
endif()

add_compile_options(-DUSE_LVGL -DCROSSPAD_REGISTER_APP_NAMED_FUNCTIONS)

# ---------------------------------------------------------------------------
# The board revision's sdkconfig, and compile-time configuration for the
# portable core (crosspad/config/*).
#
# Definitions below MUST be set before project(): ESP-IDF adds every component as
# a subdirectory from inside project(), and directory compile properties only
# propagate into subdirectories created AFTER they are set. Definitions added
# after project() land on nothing — they show up in compile_commands.json for a
# few late targets and never reach component sources.
# ---------------------------------------------------------------------------

# Every build names its board revision; ./sdkconfig belongs to no board.
if(NOT SDKCONFIG OR SDKCONFIG MATCHES "(^|[/\\\\])sdkconfig$")
    message(FATAL_ERROR
        "No board revision chosen for this build.\n"
        "  idf.py board                                        which board is connected\n"
        "  idf.py -B build_v2 -DSDKCONFIG=sdkconfig.v2 build   build for v2")
endif()

# The sdkconfig must exist and name a board revision: a missing one would be
# generated from sdkconfig.defaults and build for whatever revision that picks.
# idf.py passes SDKCONFIG as typed (sdkconfig.v2), and CMake runs in the build
# directory: resolve it against the project, as project.cmake does, or EXISTS
# looks in build_v2/.
get_filename_component(_crosspad_sdkconfig "${SDKCONFIG}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
if(NOT EXISTS "${_crosspad_sdkconfig}")
    message(FATAL_ERROR "SDKCONFIG=${SDKCONFIG}: ${_crosspad_sdkconfig} does not exist (sdkconfig.v1 / sdkconfig.v2 are checked in)")
endif()
file(READ "${_crosspad_sdkconfig}" _crosspad_sdkconfig_text)
if(NOT _crosspad_sdkconfig_text MATCHES "CONFIG_BSP_BOARD_REV_V[0-9]+=y")
    message(FATAL_ERROR "${_crosspad_sdkconfig} selects no board revision (CONFIG_BSP_BOARD_REV_V1/V2)")
endif()

# Feature-flag overrides generated by the app manager from crosspad.config.json.
# Gitignored working data carrying only flags that deviate from the checked-in
# Configuration.h defaults — an untouched project builds exactly as the headers
# say it should.
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/.crosspad/build_flags.cmake")
    include("${CMAKE_CURRENT_LIST_DIR}/.crosspad/build_flags.cmake")
    message(STATUS "CrossPad feature flags: set ${CROSSPAD_FLAGS_HASH}")
endif()

# Which firmware kit this image is (CPFI kit_id in main/fw_image_desc.c). CI
# passes the profile it applied (-DCROSSPAD_FW_KIT=lite); a plain build is
# "full"; v<digits> is taken by revision names (v1, v2) and refused. Same rule as tools/ci_matrix.py: the 16-byte CPFI field and the
# release file names (CrossPad-<tag>-<rev>-<kit>.bin) both depend on it.
# Before project() for the same reason as the blocks above.
if(DEFINED CROSSPAD_FW_KIT)
    string(LENGTH "${CROSSPAD_FW_KIT}" _crosspad_kit_len)
    if(NOT CROSSPAD_FW_KIT MATCHES "^[a-z][a-z0-9_]*$" OR _crosspad_kit_len GREATER 15
            OR CROSSPAD_FW_KIT MATCHES "^v[0-9]+$")
        message(FATAL_ERROR "CROSSPAD_FW_KIT=${CROSSPAD_FW_KIT}: a kit id is a-z, then a-z 0-9 _, at most 15 characters, not v<digits>")
    endif()
    add_compile_definitions(CROSSPAD_FW_KIT_ID="${CROSSPAD_FW_KIT}")
    message(STATUS "CrossPad firmware kit: ${CROSSPAD_FW_KIT}")
endif()

# What a kit's profile (config/profiles/[community/]<kit>.json) says about the
# image itself rather than its apps. "standalone" names the one app a
# single-app firmware starts into -- no launcher, only that app in the
# registry (main/app/CMakeLists.txt); "top_bar": false gives it the whole
# panel; "sample_engine": false leaves the sampler's 4.6 MB arena, the
# start-up sound and the kit scan out of the boot (a game needs the PSRAM);
# "radio": false hands the radio to the app -- no WiFi lease, no BLE, no
# remote session is ever started (the SDR receives through it); "audio":
# false starts no audio tasks, codecs or USB audio, whose internal RAM such
# an app needs more; "sdkconfig" lists Kconfig lines ("CONFIG_X=y", "# CONFIG_X
# is not set") laid over the revision's sdkconfig for this kit alone -- the
# result is the build dir's sdkconfig.<kit>, the checked-in file is not touched.
# Before project() for the same reason as the blocks above.
set(CROSSPAD_FW_STANDALONE "")
set(CROSSPAD_FW_TOP_BAR 1)
set(CROSSPAD_FW_SAMPLE_ENGINE 1)
set(CROSSPAD_FW_RADIO 1)
set(CROSSPAD_FW_AUDIO 1)
set(CROSSPAD_FW_KIT_APPS "")
set(CROSSPAD_FW_SDKCONFIG "")
if(DEFINED CROSSPAD_FW_KIT)
    foreach(_dir "config/profiles" "config/profiles/community")
        if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/${_dir}/${CROSSPAD_FW_KIT}.json")
            file(READ "${CMAKE_CURRENT_LIST_DIR}/${_dir}/${CROSSPAD_FW_KIT}.json" _kit_profile)
        endif()
    endforeach()
    if(DEFINED _kit_profile)
        string(JSON _v ERROR_VARIABLE _err GET "${_kit_profile}" standalone)
        if(NOT _err)
            set(CROSSPAD_FW_STANDALONE "${_v}")
        endif()
        string(JSON _v ERROR_VARIABLE _err GET "${_kit_profile}" top_bar)
        if(NOT _err AND NOT _v)
            set(CROSSPAD_FW_TOP_BAR 0)
        endif()
        string(JSON _v ERROR_VARIABLE _err GET "${_kit_profile}" sample_engine)
        if(NOT _err AND NOT _v)
            set(CROSSPAD_FW_SAMPLE_ENGINE 0)
        endif()
        string(JSON _v ERROR_VARIABLE _err GET "${_kit_profile}" radio)
        if(NOT _err AND NOT _v)
            set(CROSSPAD_FW_RADIO 0)
        endif()
        string(JSON _v ERROR_VARIABLE _err GET "${_kit_profile}" audio)
        if(NOT _err AND NOT _v)
            set(CROSSPAD_FW_AUDIO 0)
        endif()
        string(JSON _n ERROR_VARIABLE _err LENGTH "${_kit_profile}" sdkconfig)
        if(NOT _err AND _n GREATER 0)
            math(EXPR _last "${_n} - 1")
            foreach(_i RANGE ${_last})
                string(JSON _line GET "${_kit_profile}" sdkconfig ${_i})
                list(APPEND CROSSPAD_FW_SDKCONFIG "${_line}")
            endforeach()
        endif()
        string(JSON _n ERROR_VARIABLE _err LENGTH "${_kit_profile}" apps)
        if(NOT _err AND _n GREATER 0)
            math(EXPR _last "${_n} - 1")
            foreach(_i RANGE ${_last})
                string(JSON _app MEMBER "${_kit_profile}" apps ${_i})
                list(APPEND CROSSPAD_FW_KIT_APPS "${_app}")
            endforeach()
        endif()
    endif()
endif()
if(CROSSPAD_FW_SDKCONFIG)
    set(_text "${_crosspad_sdkconfig_text}")
    foreach(_line IN LISTS CROSSPAD_FW_SDKCONFIG)
        if(NOT _line MATCHES "^(# )?(CONFIG_[A-Z0-9_]+)(=.+| is not set)$")
            message(FATAL_ERROR "kit ${CROSSPAD_FW_KIT}: sdkconfig line \"${_line}\" is not CONFIG_X=value or # CONFIG_X is not set")
        endif()
        string(REGEX REPLACE "(^|\n)(# )?${CMAKE_MATCH_2}(=[^\n]*| is not set)\n" "\\1" _text "${_text}")
        string(APPEND _text "${_line}\n")
    endforeach()
    # Written only when the text changes: Kconfig rewrites the copy it reads,
    # and a copy rewritten on every configure would configure again forever.
    set(_src "${CMAKE_BINARY_DIR}/sdkconfig.${CROSSPAD_FW_KIT}.src")
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig.${CROSSPAD_FW_KIT}")
    set(_old "")
    if(EXISTS "${_src}")
        file(READ "${_src}" _old)
    endif()
    if(NOT _old STREQUAL _text OR NOT EXISTS "${SDKCONFIG}")
        file(WRITE "${_src}" "${_text}")
        file(WRITE "${SDKCONFIG}" "${_text}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_crosspad_sdkconfig}")
    list(LENGTH CROSSPAD_FW_SDKCONFIG _n)
    message(STATUS "CrossPad firmware kit: ${_n} sdkconfig line(s) over ${_crosspad_sdkconfig}")
endif()
add_compile_definitions(
    CROSSPAD_FW_STANDALONE_APP="${CROSSPAD_FW_STANDALONE}"
    CROSSPAD_FW_TOP_BAR=${CROSSPAD_FW_TOP_BAR}
    CROSSPAD_FW_SAMPLE_ENGINE=${CROSSPAD_FW_SAMPLE_ENGINE}
    CROSSPAD_FW_RADIO=${CROSSPAD_FW_RADIO}
    CROSSPAD_FW_AUDIO=${CROSSPAD_FW_AUDIO})
if(CROSSPAD_FW_STANDALONE)
    message(STATUS "CrossPad firmware kit: single app ${CROSSPAD_FW_STANDALONE}, top bar ${CROSSPAD_FW_TOP_BAR}, sample engine ${CROSSPAD_FW_SAMPLE_ENGINE}, radio ${CROSSPAD_FW_RADIO}, audio ${CROSSPAD_FW_AUDIO}")
endif()

# The only host a CrossPad sends its crosspad.app device token to
# (docs/design/2026-10-07-account-sync-design.md). Compile-time on purpose: FW_UPD_URL and
# anything else set at runtime can never receive the token. Test images point
# it at a local test server or the dev Worker; ACCT_STATUS prints it. Before
# project() for the same reason as the blocks above.
set(CROSSPAD_ACCOUNT_URL "https://crosspad.app" CACHE STRING "crosspad.app base URL for device pairing")
if(NOT CROSSPAD_ACCOUNT_URL MATCHES "^https?://[A-Za-z0-9.-]+(:[0-9]+)?$")
    message(FATAL_ERROR "CROSSPAD_ACCOUNT_URL=${CROSSPAD_ACCOUNT_URL}: http(s)://host[:port], no path")
endif()
add_compile_definitions(CROSSPAD_ACCOUNT_URL="${CROSSPAD_ACCOUNT_URL}")
# Where the owner types the code: the account app, not the device API's host
# (crosspad.app serves only /api/device/* from the Worker; accounts live on play.).
set(CROSSPAD_LINK_PAGE "play.crosspad.app/device/link" CACHE STRING "page the pairing screen sends the owner to")
add_compile_definitions(CROSSPAD_LINK_PAGE="${CROSSPAD_LINK_PAGE}")
if(NOT CROSSPAD_ACCOUNT_URL STREQUAL "https://crosspad.app")
    message(WARNING "CrossPad account URL: ${CROSSPAD_ACCOUNT_URL} -- a bench image, not for release")
endif()
