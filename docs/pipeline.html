  %% -------- INPUT → CORE --------
  STM -->|SysEx| SMH
  SMH -->|handlePadPress idx,vel| PM
  ENC -->|encoder counter| SMH
  SMH -.->|input.encoderTop| MIH
  USB -->|handleNoteOn| MIH
  BLE -->|handleNoteOn| MIH

  %% -------- CORE forward --------
  PM -->|onPadPress| SPL
  PM -->|onPadPress| QPL
  PM ==>|postNoteOn + postPadPressed
src=PadManager| BUS
  PM -->|sendNoteOn src=PadMgr| MOUT
  MIH -->|handleMidiNoteOn| PM
  MIH ==>|postNoteOn src=MidiInput| BUS
  MIH -->|sendNoteOn/CC| MOUT

  %% -------- SAMPLER forward --------
  SPL ==>|postSamplePlayback on=true
src=App| BUS
  SPL -->|trigger voice| SENG
  KIT -.->|pad colors onActivate| SPL
  SENG -->|render frames| SNODE
  SNODE -->|audio frames| MIX
  MIX --> DAC

  %% -------- SEQUENCER forward --------
  QPL -->|transport / record| QCTL
  QCTL -->|setBpm / play / stop| QENG
  QENG -->|step tick → NoteOn cb| QEVH
  QPL ==>|live: postNoteOn src=App
rec: src=SequencerInput| BUS
  QEVH ==>|step NoteOn
src=SequencerCallback| BUS

  %% =================================================
  %% ============ FEEDBACK LOOPS =====================
  %% =================================================

  %% FB-LED: SamplePlayback → LED
  BUS -.->|SamplePlayback → notifyNotePlay| LED
  PM -.->|notifyNoteOn vel→kolor| LED
  LED -.->|setPixel + refresh| STRIP

  %% FB sampler voice end → LED off
  SENG -.->|engine_note_off → handleSamplePlay off| PM
  SPL -.->|hold mode: onPadRelease
postSamplePlayback off| BUS

  %% FB-1 MIDI-in świeci pady
  MIH -.->|MIDI-in → pad LED| LED

  %% FB-2 / FB-3 MIDI routing back
  MOUT -.->|send2STM → MIDI back| STM
  MIH -.->|BLE2Midi / Midi2BLE most| MOUT

  %% FB seq → LED animacja kroków
  QEVH -.->|barChanged → playPadAnimation| ANI
  ANI -.->|frames| STRIP
  QPL -.->|updateTransportColors
pad12 rec/14 play/15 click| LED

  %% FB seq NoteOn → LED wizual + (chain do samplera via MIDI/EventBus)
  BUS -.->|NoteOn → LED wizual| LED
  BUS -.->|NoteOn src=SeqCallback
→ MIDI out → loop→ sampler audio| MOUT
